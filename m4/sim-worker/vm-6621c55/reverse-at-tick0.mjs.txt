import { chromium } from "playwright-core";
import { writeFileSync } from "node:fs";
const [url, out] = process.argv.slice(2);
const browser = await chromium.launch({ executablePath: process.env.CHROME ?? "/usr/local/bin/google-chrome", headless: false, args: ["--ignore-gpu-blocklist"] });
const page = await browser.newPage({ viewport: { width: 1280, height: 720 } });
const errors = [];
page.on("pageerror", (e) => errors.push(e.message));
await page.goto(url);
await page.waitForSelector("#sim", { state: "attached", timeout: 120000 });
await page.waitForTimeout(3000);
const before = JSON.parse(await page.locator("#sim").textContent());
const hudBefore = await page.locator(".hud").innerText();
await page.screenshot({ path: `${out}-before.png` });
await page.keyboard.press("r");
await page.waitForTimeout(2000);
const afterText = await page.locator("#sim").textContent({ timeout: 1000 }).catch(() => null);
const after = afterText ? JSON.parse(afterText) : { error: await page.locator(".hud .error").textContent({ timeout: 1000 }).catch(() => null) };
const hudAfter = await page.locator(".hud").innerText();
await page.screenshot({ path: `${out}-after.png` });
for (const key of ["5", "6", "7", "8", "9"]) await page.keyboard.press(key);
await page.waitForTimeout(1500);
const hudError = await page.locator(".hud .error").textContent({ timeout: 500 }).catch(() => null);
writeFileSync(`${out}.json`, JSON.stringify({ url, before, after, errors, hudError, hudBefore, hudAfter }, null, 1));
console.log(JSON.stringify({ before: { tick: before.tick, edge: before.trainEdge, offset: before.trainOffsetLdu, bogies: before.bogiePointsLdu }, after: after.error ? after : { tick: after.tick, edge: after.trainEdge, offset: after.trainOffsetLdu, bogies: after.bogiePointsLdu }, errors, hudError }));
await browser.close();
