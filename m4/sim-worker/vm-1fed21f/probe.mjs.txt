import { chromium } from "playwright-core";
const [url, out, waitMs] = process.argv.slice(2);
const browser = await chromium.launch({ executablePath: process.env.CHROME ?? "/usr/local/bin/google-chrome", headless: false, args: ["--enable-unsafe-webgpu", "--ignore-gpu-blocklist", "--window-size=1600,900"] });
const page = await browser.newPage({ viewport: { width: Number(process.env.W ?? 1600), height: Number(process.env.H ?? 900) } });
page.on("console", (m) => console.log("console", m.type(), m.text()));
page.on("pageerror", (e) => console.log("pageerror", e.message));
await page.goto(url);
await page.waitForTimeout(Number(waitMs ?? 8000));
console.log(await page.locator(".hud").innerText().catch(() => ""));
console.log("BENCH", await page.locator("#sim-bench").textContent({ timeout: 1000 }).catch(() => "none"));
console.log("SIM", await page.locator("#sim").textContent({ timeout: 1000 }).catch(() => "none"));
await page.screenshot({ path: out });
await browser.close();
